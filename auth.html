/* ========================================================= GOOGLE ROLE SETUP New Google users should NOT automatically become student. Email/password users keep their existing student/owner flow. ========================================================= */ /* --------------------------------------------------------- 1. handle_new_user() --------------------------------------------------------- */ CREATE OR REPLACE FUNCTION public.handle_new_user() RETURNS trigger LANGUAGE plpgsql SECURITY DEFINER SET search_path TO 'public' AS $function$ DECLARE requested_role text; auth_provider text; BEGIN auth_provider := lower( coalesce( new.raw_app_meta_data->>'provider', new.raw_user_meta_data->>'provider', '' ) ); requested_role := lower( trim( coalesce( new.raw_user_meta_data->>'role', '' ) ) ); /* Normal email/password signup: role comes from signup form. Google signup: role stays NULL until auth.html saves the Student/Owner selection. */ IF requested_role NOT IN ('student', 'owner') THEN IF auth_provider = 'google' THEN requested_role := NULL; ELSE requested_role := 'student'; END IF; END IF; INSERT INTO public.profiles ( id, full_name, phone, email, role ) VALUES ( new.id, nullif( trim( coalesce( new.raw_user_meta_data->>'full_name', '' ) ), '' ), nullif( trim( coalesce( new.raw_user_meta_data->>'phone', '' ) ), '' ), new.email, requested_role ) ON CONFLICT (id) DO UPDATE SET email = excluded.email, full_name = coalesce( nullif(trim(public.profiles.full_name), ''), excluded.full_name ), phone = coalesce( nullif(trim(public.profiles.phone), ''), excluded.phone ); RETURN new; END; $function$; /* --------------------------------------------------------- 2. sync_profile_from_auth() --------------------------------------------------------- */ CREATE OR REPLACE FUNCTION public.sync_profile_from_auth() RETURNS trigger LANGUAGE plpgsql SECURITY DEFINER SET search_path TO 'public' AS $function$ DECLARE requested_role text; auth_provider text; BEGIN auth_provider := lower( coalesce( NEW.raw_app_meta_data->>'provider', NEW.raw_user_meta_data->>'provider', '' ) ); requested_role := lower( trim( coalesce( NEW.raw_user_meta_data->>'role', '' ) ) ); /* Email/password: use selected role. Google: leave role NULL until auth.html saves Student/Owner choice. */ IF requested_role NOT IN ('student', 'owner') THEN IF auth_provider = 'google' THEN requested_role := NULL; ELSE requested_role := 'student'; END IF; END IF; INSERT INTO public.profiles ( id, full_name, phone, role, email ) VALUES ( NEW.id, COALESCE( NEW.raw_user_meta_data->>'full_name', split_part( COALESCE(NEW.email, ''), '@', 1 ), 'User' ), COALESCE( NEW.raw_user_meta_data->>'phone', '' ), requested_role, NEW.email ) ON CONFLICT (id) DO UPDATE SET email = EXCLUDED.email; RETURN NEW; END; $function$;
